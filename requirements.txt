#
#    For a CPU-only install instead:
#
#        python -m pip install torch==2.8.0 \
#            --index-url https://download.pytorch.org/whl/cpu
#
# 4. From the repo root (`code_for_paper/`):
#
#        python -m pip install -r requirements.txt
#
# 5. (Optional) Pre-cache the public TabICL classifier checkpoint so the
#    HPO flow does not need internet at run time:
#
#        python -c "from huggingface_hub import hf_hub_download; \
